<!-- Interactive Scripts -->
<script>
let selectedColor = '#6366f1';

function setTheme(color) {
selectedColor = color;
document.getElementById('previewTitle').style.color = color;
document.getElementById('previewBtn').style.backgroundColor = color;
document.getElementById('previewCta').style.backgroundColor = color;
}

function generateDemo() {
const name = document.getElementById('bizName').value || 'My Business';
const cat = document.getElementById('bizCategory').value;

document.getElementById('previewTitle').innerText = name;

if(cat === 'Bakery & Cafe') {
document.getElementById('previewHeading').innerText = "Freshly Baked Delights Every Day";
document.getElementById('previewSub').innerText = "Welcome to " + name + ". We serve premium artisanal breads, cakes, and hand-crafted beverages.";
} else if(cat === 'Fitness Gym') {
document.getElementById('previewHeading').innerText = "Transform Your Body & Mind";
document.getElementById('previewSub').innerText = "Join " + name + " today for state-of-the-art equipment, expert personal trainers, and high-energy classes.";
} else if(cat === 'Hair & Beauty Salon') {
document.getElementById('previewHeading').innerText = "Style & Luxury You Deserve";
document.getElementById('previewSub').innerText = "At " + name + ", we specialize in modern haircuts, coloring, skincare, and rejuvenating spa treatments.";
} else if(cat === 'Tech Startup') {
document.getElementById('previewHeading').innerText = "Next-Gen Digital Solutions";
document.getElementById('previewSub').innerText = name + " delivers cutting-edge software and automation tools built to scale your enterprise.";
}
}

function updateEarnings() {
const clients = parseInt(document.getElementById('clientSlider').value);
const fee = parseInt(document.getElementById('feeSlider').value);

document.getElementById('clientCountText').innerText = clients + " Clients";
document.getElementById('feeText').innerText = "₹" + fee.toLocaleString('en-IN');

const monthly = clients * fee;
const yearly = monthly * 12;

document.getElementById('monthlyEarnings').innerText = "₹" + monthly.toLocaleString('en-IN');
document.getElementById('yearlyEarnings').innerText = "₹" + yearly.toLocaleString('en-IN');
}
</script>
